    -H "Authorization: Bearer eyJhbGciOiJIUzI1Ni…" \
    -H "Content-Type: application/json" -d '{"title":""}'
HTTP 400
{"error":"Datos de entrada inválidos.","detalles":[{"campo":"title","mensaje":"El título debe tener entre 3 y 80 caracteres."},{"campo":"category","mensaje":"La categoría debe ser una de: Infraestructura, Aplicación, Seguridad, Acceso, Otro."},{"campo":"priority","mensaje":"La prioridad debe ser una de: Alta, Media, Baja."}]}

# Mass assignment: campo role no permitido
$ curl -i -X POST http://localhost:3000/api/incidentes \
    -H "Authorization: Bearer eyJhbGciOiJIUzI1Ni…" \
    -H "Content-Type: application/json" -d '{"title":"Intento","category":"Otro","priority":"Baja","role":"admin"}'
HTTP 400
{"error":"Datos de entrada inválidos.","detalles":[{"campo":"role","mensaje":"Campo no permitido."}]}
