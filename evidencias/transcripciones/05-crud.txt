# 1) CREAR (analista)
$ curl -i -X POST http://localhost:3000/api/incidentes \
    -H "Authorization: Bearer eyJhbGciOiJIUzI1Ni…" \
    -H "Content-Type: application/json" -d '{"title":"Caída del servicio de correo","category":"Infraestructura","priority":"Alta","email":"ana@incidenta.ec","description":"No se envían correos desde las 08:00."}'
HTTP 201
{"id":6,"title":"Caída del servicio de correo","category":"Infraestructura","priority":"Alta","status":"Abierto","date":"2026-10-21","email":"ana@incidenta.ec","description":"No se envían correos desde las 08:00.","created_at":"2026-10-21T04:16:19.267Z","updated_at":"2026-10-21T04:16:19.267Z"}

# 2) CONSULTAR (lector)
$ curl -i -X GET http://localhost:3000/api/incidentes/6 \
    -H "Authorization: Bearer eyJhbGciOiJIUzI1Ni…"
HTTP 200
{"id":6,"title":"Caída del servicio de correo","category":"Infraestructura","priority":"Alta","status":"Abierto","date":"2026-10-21","email":"ana@incidenta.ec","description":"No se envían correos desde las 08:00.","created_at":"2026-10-21T04:16:19.267Z","updated_at":"2026-10-21T04:16:19.267Z"}

# 3) ACTUALIZAR (analista)
$ curl -i -X PATCH http://localhost:3000/api/incidentes/6 \
    -H "Authorization: Bearer eyJhbGciOiJIUzI1Ni…" \
    -H "Content-Type: application/json" -d '{"status":"Resuelto"}'
HTTP 200
{"id":6,"title":"Caída del servicio de correo","category":"Infraestructura","priority":"Alta","status":"Resuelto","date":"2026-10-21","email":"ana@incidenta.ec","description":"No se envían correos desde las 08:00.","created_at":"2026-10-21T04:16:19.267Z","updated_at":"2026-10-21T04:16:19.271Z"}

$ curl -i -X DELETE http://localhost:3000/api/incidentes/6 \
    -H "Authorization: Bearer eyJhbGciOiJIUzI1Ni…"

(sin cuerpo)
